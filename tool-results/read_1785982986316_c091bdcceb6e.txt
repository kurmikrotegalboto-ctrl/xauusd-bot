    20→function fmtMoney(v: number | null | undefined): string {
    21→  if (v == null || !Number.isFinite(v)) return '$0.00'
    22→  const sign = v < 0 ? '-' : ''
    23→  return `${sign}$${Math.abs(v).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    24→}
    25→
    26→function fmtPrice(v: number | null | undefined): string {
    27→  if (v == null || !Number.isFinite(v)) return '0.00'
    28→  return v.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })
    29→}
    30→
    31→function fmtPct(v: number | null | undefined): string {
    32→  if (v == null || !Number.isFinite(v)) return '+0.00%'
    33→  return `${v >= 0 ? '+' : ''}${v.toFixed(2)}%`
    34→}
    35→
    36→function fmtNum(v: number | null | undefined, digits = 2): string {
    37→  if (v == null || !Number.isFinite(v)) return '0.00'
    38→  return v.toFixed(digits)
    39→}
    40→
    41→function fmtTime(ts: number): string {
    42→  return new Date(ts).toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit', second: '2-digit' })
    43→}
    44→
    45→function fmtDuration(ms: number): string {
    46→  const sec = Math.floor(ms / 1000)
    47→  if (sec < 60) return `${sec}s`
    48→  const min = Math.floor(sec / 60)
    49→  if (min < 60) return `${min}m ${sec % 60}s`
    50→  const hr = Math.floor(min / 60)
    51→  return `${hr}h ${min % 60}m`
    52→}
    53→
    54→function exitReasonBadge(reason: ExitReason | null) {
    55→  switch (reason) {
    56→    case 'TP':
    57→      return <Badge className="bg-emerald-500/15 text-emerald-400 border-emerald-500/30 hover:bg-emerald-500/20">TP HIT</Badge>
    58→    case 'SL':
    59→      return <Badge className="bg-rose-500/15 text-rose-400 border-rose-500/30 hover:bg-rose-500/20">SL HIT</Badge>
    60→    case 'EXPIRED':
    61→      return <Badge className="bg-amber-500/15 text-amber-400 border-amber-500/30 hover:bg-amber-500/20">EXPIRED</Badge>
    62→    case 'MANUAL':
    63→      return <Badge className="bg-sky-500/15 text-sky-400 border-sky-500/30 hover:bg-sky-500/20">MANUAL</Badge>
    64→    default:
    65→      return <Badge variant="outline">—</Badge>
    66→  }
    67→}

   631→function OpenPositionRow({
   632→  p, currentPrice, onClose, busy,
   633→}: {
   634→  p: Position
   635→  currentPrice: number
   636→  onClose: () => void
   637→  busy: boolean
   638→}) {
   639→  const diff = p.side === 'BUY' ? currentPrice - p.entryPrice : p.entryPrice - currentPrice
   640→  const pnl = diff * (p.lotSize ?? 0) * 100
   641→  const rPct = p.riskAmount > 0 ? (pnl / p.riskAmount) * 100 : 0
   642→  const ageMs = Date.now() - (p.openTime ?? Date.now())
   643→  const ageMin = Math.floor(ageMs / 60000)
   644→  const ageSec = Math.floor((ageMs % 60000) / 1000)
   645→
   646→  const distToSl = Math.abs(currentPrice - (p.stopLoss ?? currentPrice))
   647→  const distToTp = Math.abs(currentPrice - (p.takeProfit ?? currentPrice))
   648→  const slDenom = Math.abs((p.entryPrice ?? 0) - (p.stopLoss ?? 0))
   649→  const tpDenom = Math.abs((p.entryPrice ?? 0) - (p.takeProfit ?? 0))
   650→
   651→  return (
   652→    <div className="rounded-lg border border-muted-foreground/15 bg-background/40 p-3">
   653→      <div className="flex items-start justify-between gap-2">
   654→        <div className="flex-1">
   655→          <div className="flex items-center gap-2">
   656→            <Badge
   657→              variant="outline"
   658→              className={
   659→                p.side === 'BUY'
   660→                  ? 'border-emerald-500/30 text-emerald-400'
   661→                  : 'border-rose-500/30 text-rose-400'
   662→              }
   663→            >
   664→              {p.side} {fmtNum(p.lotSize, 2)} lot
   665→            </Badge>
   666→            <span className="text-[10px] text-muted-foreground">
   667→              Conf {p.confidence}% • {ageMin}m {ageSec}s
   668→            </span>
   669→          </div>
   670→          <div className="mt-1.5 grid grid-cols-4 gap-2 text-[10px] font-mono">
   671→            <div>
   672→              <div className="text-muted-foreground">Entry</div>
   673→              <div>{fmtPrice(p.entryPrice)}</div>
   674→            </div>
   675→            <div>
   676→              <div className="text-muted-foreground">SL</div>
   677→              <div className="text-rose-400/80">{fmtPrice(p.stopLoss)}</div>
   678→            </div>
   679→            <div>
   680→              <div className="text-muted-foreground">TP</div>
   681→              <div className="text-emerald-400/80">{fmtPrice(p.takeProfit)}</div>
   682→            </div>
   683→            <div>
   684→              <div className="text-muted-foreground">Cur</div>
   685→              <div>{fmtPrice(currentPrice)}</div>
   686→            </div>
   687→          </div>
   688→          <div className="mt-2 flex items-center gap-3 text-[10px]">
   689→            <span className="text-muted-foreground">
   690→              → SL: {fmtPrice(distToSl)} ({slDenom > 0 ? fmtNum((distToSl / slDenom) * 100, 0) : '0'}%)
   691→            </span>
   692→            <span className="text-muted-foreground">
   693→              → TP: {fmtPrice(distToTp)} ({tpDenom > 0 ? fmtNum((distToTp / tpDenom) * 100, 0) : '0'}%)
   694→            </span>
   695→          </div>
   696→        </div>
   697→        <div className="text-right">
   698→          <div className={`font-mono text-sm font-bold ${pnl >= 0 ? 'text-emerald-400' : 'text-rose-400'}`}>
   699→            {fmtMoney(pnl)}
   700→          </div>
   701→          <div className={`font-mono text-[10px] ${rPct >= 0 ? 'text-emerald-400' : 'text-rose-400'}`}>
   702→            {fmtPct(rPct)}
   703→          </div>
   704→          <Button
   705→            size="sm"
   706→            variant="ghost"
   707→            onClick={onClose}
   708→            disabled={busy}
   709→            className="mt-1 h-6 px-2 text-[10px] text-muted-foreground hover:text-foreground"
   710→          >
   711→            Close
   712→          </Button>
   713→        </div>
   714→      </div>
   715→    </div>
   716→  )
   717→}
